using System;

namespace WinFormsApp2
{
    class Triangle
    {
        public double a;
        public double b;

        public double Gipotenuza()
        {
            return Math.Sqrt(a * a + b * b);
        }

        public double Perimeter()
        {
            return a + b + Gipotenuza();
        }

        public double Area()
        {
            return a * b / 2;
        }

        public string Info()
        {
            return "Прямокутний трикутник: a = " + a + ", b = " + b + ", c = " + Math.Round(Gipotenuza(), 2) + "\r\n" +
                   "  Периметр = " + Math.Round(Perimeter(), 2) + "\r\n" +
                   "  Площа = " + Math.Round(Area(), 2) + "\r\n";
        }

        public static Triangle operator ++(Triangle t)
        {
            return new Triangle { a = t.a + 1, b = t.b + 1 };
        }

        public static Triangle operator --(Triangle t)
        {
            return new Triangle { a = t.a - 1, b = t.b - 1 };
        }
    }

    class Rectangle : Triangle
    {
        public new double Perimeter()
        {
            return 2 * (a + b);
        }

        public new double Area()
        {
            return a * b;
        }

        public new string Info()
        {
            return "Прямокутник: a = " + a + ", b = " + b + "\r\n" +
                   "  Периметр = " + Math.Round(Perimeter(), 2) + "\r\n" +
                   "  Площа = " + Math.Round(Area(), 2) + "\r\n";
        }

        public static Rectangle operator ++(Rectangle r)
        {
            return new Rectangle { a = r.a + 1, b = r.b + 1 };
        }

        public static Rectangle operator --(Rectangle r)
        {
            return new Rectangle { a = r.a - 1, b = r.b - 1 };
        }
    }
}

namespace WinFormsApp2
{
    public partial class Form1 : Form
    {
        Triangle t = new Triangle { a = 3, b = 4 };
        Rectangle r = new Rectangle { a = 2, b = 5 };

        public Form1()
        {
            InitializeComponent();
            ShowInfo();
        }

        void ShowInfo()
        {
            textBox.Text = t.Info() + "\r\n" + r.Info();
        }

        private void buttonT1_Click(object? sender, EventArgs e)
        {
            t++;
            ShowInfo();
        }

        private void buttonT2_Click(object? sender, EventArgs e)
        {
            t--;
            ShowInfo();
        }

        private void buttonR1_Click(object? sender, EventArgs e)
        {
            r++;
            ShowInfo();
        }

        private void buttonR2_Click(object? sender, EventArgs e)
        {
            r--;
            ShowInfo();
        }
    }
}
