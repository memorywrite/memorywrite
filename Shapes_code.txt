using System;

namespace ShapesApp
{
    class RightTriangle
    {
        public double a;
        public double b;

        public RightTriangle(double x, double y)
        {
            a = x;
            b = y;
        }

        public double Gipotenuza()
        {
            return Math.Sqrt(a * a + b * b);
        }

        public double Perimeter()
        {
            return a + b + Gipotenuza();
        }

        public double Area()
        {
            return a * b / 2;
        }

        public void Show()
        {
            Console.WriteLine("Прямокутний трикутник: a = " + a + ", b = " + b +
                              ", c = " + Math.Round(Gipotenuza(), 2));
            Console.WriteLine("  Периметр = " + Math.Round(Perimeter(), 2));
            Console.WriteLine("  Площа    = " + Math.Round(Area(), 2));
        }
    }

    class Rectangle : RightTriangle
    {
        public Rectangle(double x, double y) : base(x, y)
        {
        }

        public new double Perimeter()
        {
            return 2 * (a + b);
        }

        public new double Area()
        {
            return a * b;
        }

        public new void Show()
        {
            Console.WriteLine("Прямокутник: a = " + a + ", b = " + b);
            Console.WriteLine("  Периметр = " + Math.Round(Perimeter(), 2));
            Console.WriteLine("  Площа    = " + Math.Round(Area(), 2));
        }
    }
}

using System;
using System.Text;

namespace ShapesApp
{
    class Program
    {
        static void Main(string[] args)
        {
            Console.OutputEncoding = Encoding.UTF8;

            RightTriangle[] shapes = new RightTriangle[5];
            shapes[0] = new RightTriangle(3, 4);
            shapes[1] = new RightTriangle(6, 8);
            shapes[2] = new Rectangle(2, 5);
            shapes[3] = new Rectangle(4, 4);
            shapes[4] = new Rectangle(7, 3);

            Console.WriteLine("Інформація про фігури:");
            Console.WriteLine();

            for (int i = 0; i < shapes.Length; i++)
            {
                Console.Write((i + 1) + ". ");
                shapes[i].Show();
                Console.WriteLine();
            }

            Console.ReadKey();
        }
    }
}
