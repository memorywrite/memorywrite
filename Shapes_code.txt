using System;

namespace ConsoleApp3
{
    class Triangle
    {
        public double a;
        public double b;

        public double Gipotenuza()
        {
            return Math.Sqrt(a * a + b * b);
        }

        public double Perimeter()
        {
            return a + b + Gipotenuza();
        }

        public double Area()
        {
            return a * b / 2;
        }

        public void Show()
        {
            Console.WriteLine("Прямокутний трикутник: a = " + a + ", b = " + b +
                              ", c = " + Math.Round(Gipotenuza(), 2));
            Console.WriteLine("  Периметр = " + Math.Round(Perimeter(), 2));
            Console.WriteLine("  Площа    = " + Math.Round(Area(), 2));
        }
    }

    class Rectangle : Triangle
    {
        public new double Perimeter()
        {
            return 2 * (a + b);
        }

        public new double Area()
        {
            return a * b;
        }

        public new void Show()
        {
            Console.WriteLine("Прямокутник: a = " + a + ", b = " + b);
            Console.WriteLine("  Периметр = " + Math.Round(Perimeter(), 2));
            Console.WriteLine("  Площа    = " + Math.Round(Area(), 2));
        }
    }
}

using System;
using System.Text;

namespace ConsoleApp3
{
    class Program
    {
        static void Main(string[] args)
        {
            Console.OutputEncoding = Encoding.UTF8;

            Triangle[] arr = new Triangle[5];
            arr[0] = new Triangle { a = 3, b = 4 };
            arr[1] = new Triangle { a = 6, b = 8 };
            arr[2] = new Rectangle { a = 2, b = 5 };
            arr[3] = new Rectangle { a = 4, b = 4 };
            arr[4] = new Rectangle { a = 7, b = 3 };

            Console.WriteLine("Інформація про фігури:");
            Console.WriteLine();

            for (int i = 0; i < arr.Length; i++)
            {
                Console.Write((i + 1) + ". ");
                arr[i].Show();
                Console.WriteLine();
            }

            Console.ReadKey();
        }
    }
}
